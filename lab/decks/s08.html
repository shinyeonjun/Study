절편 포함한 최소제곱

import numpy as np
x, y = np.array([0, 1, 2]), np.array([1, 2, 5])
X = np.column_stack((np.ones(3), x))
coef = np.linalg.lstsq(X, y, rcond=None)[0]
print(coef.round(4).tolist())
print(f"{np.square(y-X@coef).sum():.6f}")
[0.6667, 2.0]
0.666667
