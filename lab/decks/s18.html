해석 경사와 중심차분

import numpy as np
x, w, y = 2., 3., 5.
loss = lambda w: (w*x + 1 - y)**2
eps = 1e-5
analytic = 2*(w*x+1-y)*x
numeric = (loss(w+eps)-loss(w-eps))/(2*eps)
assert np.isclose(analytic,numeric,rtol=0,atol=1e-8)
print(f"{loss(w):.0f} {analytic:.0f} {numeric:.4f}")
