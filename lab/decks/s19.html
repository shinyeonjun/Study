PyTorch로 한 가중치 세 번 갱신

import torch
x = torch.tensor([[1.], [2.]])
y = 2*x
model = torch.nn.Linear(1, 1, bias=False)
with torch.no_grad():
    model.weight.zero_()
opt = torch.optim.SGD(model.parameters(), lr=.1)
model.train()
for _ in range(3):
    opt.zero_grad()
    loss = (model(x)-y).square().mean()
    loss.backward()
    opt.step()
model.eval()
with torch.no_grad():
    print(f"{model.weight.item():.2f} {(model(x)-y).square().mean().item():.5f}")
