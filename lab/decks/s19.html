assert 예측.shape == 정답.shape = '이래야 한다'를 코드로버전 = Python·numpy 번호이상하면 작은 배치를 손계산과 대조

배치 고리를 numpy로

import numpy as np
x = np.array([-2., -1., 0., 1., 2.]); y = 2 * x + 1
w, b, lr = 0.0, 0.0, 0.1
rng = np.random.default_rng(7)                     # 시드 7
for epoch in range(8):
    total, count = 0.0, 0
    for idx in np.split(rng.permutation(5), [2, 4]):   # 배치 2, 2, 1
        err = (w * x[idx] + b) - y[idx]               # 예측 − 정답
        assert err.shape == y[idx].shape
        gw = 2 * np.mean(err * x[idx])                # 기울기를 새로 계산
        gb = 2 * np.mean(err)
        w, b = w - lr * gw, b - lr * gb               # 한 발
        total += np.sum(err ** 2); count += len(idx)  # 합과 개수
    if epoch in (0, 7):
        print(epoch + 1, round(w, 4), round(b, 4), round(total / count, 4))
1 1.7 0.46 5.6
8 2.001 0.9963 0.0001
