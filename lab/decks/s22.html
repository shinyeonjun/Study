16가지 부호로 양쪽 꼬리 계산

from itertools import product
d = [1,2,3,4]
sums = [sum(v*b for v,b in zip(d,sign))
        for sign in product([-1,1],repeat=4)]
extreme = sum(abs(v)>=abs(sum(d)) for v in sums)
print(sum(d)/len(d), extreme, len(sums), extreme/len(sums))
2.5 2 16 0.125
